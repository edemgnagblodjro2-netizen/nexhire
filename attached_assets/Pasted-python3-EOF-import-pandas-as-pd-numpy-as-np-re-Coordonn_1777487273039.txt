def extract_postal(addr):
    if pd.isna(addr): return None
    m = re.search(r'\b([A-Za-z]\d[A-Za-z])\s*(\d[A-Za-z]\d)\b', str(addr))
    if m: return f"{m.group(1).upper()} {m.group(2).upper()}"
    return None

geocoded = 0
city_fallback = 0
already = 0
no_match = 0

for idx, row in df.iterrows():
    # Déjà GPS → skip
    if pd.notna(row['lat']) and pd.notna(row['lng']):
        already += 1
        continue
    
    # Sans adresse → on fait rien
    if pd.isna(row['address']):
        continue
    
    # Essai 1: par code postal
    postal = extract_postal(row['address'])
    if postal and postal in postal_gps:
        df.at[idx, 'lat'] = round(postal_gps[postal][0], 6)
        df.at[idx, 'lng'] = round(postal_gps[postal][1], 6)
        df.at[idx, 'sans_gps'] = np.nan
        geocoded += 1
        continue
    
    # Essai 2: par ville (fallback approximatif)
    city = str(row['city']).strip() if pd.notna(row['city']) else ''
    if city in ville_coords:
        # Légère variation aléatoire déterministe basée sur l'index pour éviter superposition
        base_lat, base_lng = ville_coords[city]
        offset = (idx % 100) * 0.0002 - 0.01  # offset ±0.01° ~1km
        df.at[idx, 'lat'] = round(base_lat + offset, 6)
        df.at[idx, 'lng'] = round(base_lng + offset * 0.8, 6)
        df.at[idx, 'sans_gps'] = np.nan
        city_fallback += 1
        continue
    
    no_match += 1

print(f"Résultats géocodage:")
print(f"  Déjà géocodés    : {already}")
print(f"  Par code postal  : {geocoded} (précision ~200m)")
print(f"  Par ville        : {city_fallback} (précision ~1km, approximatif)")
print(f"  Non géocodés     : {no_match}")
print(f"\nTotal avec GPS maintenant: {df['lat'].notna().sum()} / {len(df)}")
print(f"Total sans GPS           : {df['lat'].isna().sum()}")
