ÉTAPE 1 — BASE DE DONNÉES
────────────────────────────────────────

Créer ces 3 tables dans PostgreSQL :

CREATE TABLE nh_posts (
  id                 SERIAL PRIMARY KEY,
  author_id          INTEGER REFERENCES nh_users(id) ON DELETE CASCADE,
  type               VARCHAR(20) NOT NULL DEFAULT 'text',
  content            TEXT,
  image_url          TEXT,
  article_title      TEXT,
  article_body       TEXT,
  job_id             INTEGER REFERENCES nh_jobs(id) ON DELETE SET NULL,
  likes_count        INTEGER DEFAULT 0,
  comments_count     INTEGER DEFAULT 0,
  shared_to_linkedin BOOLEAN DEFAULT FALSE,
  linkedin_post_id   TEXT,
  is_active          BOOLEAN DEFAULT TRUE,
  created_at         TIMESTAMP DEFAULT NOW(),
  updated_at         TIMESTAMP DEFAULT NOW()
);

CREATE TABLE nh_post_likes (
  id         SERIAL PRIMARY KEY,
  post_id    INTEGER REFERENCES nh_posts(id) ON DELETE CASCADE,
  user_id    INTEGER REFERENCES nh_users(id) ON DELETE CASCADE,
  created_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(post_id, user_id)
);

CREATE TABLE nh_post_comments (
  id         SERIAL PRIMARY KEY,
  post_id    INTEGER REFERENCES nh_posts(id) ON DELETE CASCADE,
  author_id  INTEGER REFERENCES nh_users(id) ON DELETE CASCADE,
  content    TEXT NOT NULL,
  likes_count INTEGER DEFAULT 0,
  parent_id  INTEGER REFERENCES nh_post_comments(id),
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE nh_job_questions (
  id          SERIAL PRIMARY KEY,
  job_id      INTEGER REFERENCES nh_jobs(id) ON DELETE CASCADE,
  author_id   INTEGER REFERENCES nh_users(id) ON DELETE CASCADE,
  question    TEXT NOT NULL,
  answer      TEXT,
  answered_by INTEGER REFERENCES nh_users(id),
  answered_at TIMESTAMP,
  is_public   BOOLEAN DEFAULT TRUE,
  created_at  TIMESTAMP DEFAULT NOW()
);

Ajouter les index de performance :
CREATE INDEX ON nh_posts(author_id, created_at DESC);
CREATE INDEX ON nh_posts(type, created_at DESC);
CREATE INDEX ON nh_post_comments(post_id, created_at);
CREATE INDEX ON nh_post_likes(post_id, user_id);
CREATE INDEX ON nh_job_questions(job_id, is_public);

ÉTAPE 2 — ROUTES API dans server.js
────────────────────────────────────────

-- FEED --
GET    /api/feed
       → Liste paginée des posts
       → Query params : ?page=1&limit=20&type=all
       → Inclure : auteur, likes_count, 
         comments_count, is_liked_by_me
       → ORDER BY created_at DESC

POST   /api/feed
       → Créer un post (auth requis)
       → Body : { type, content, image_url,
                  article_title, article_body, job_id }
       → Valider : content obligatoire si type=text
       → Valider : article_title + article_body 
                   obligatoires si type=article

GET    /api/feed/:id
       → Un post avec ses commentaires

DELETE /api/feed/:id
       → Seulement l'auteur peut supprimer
       → Soft delete : is_active = false

-- LIKES --
POST   /api/feed/:id/like
       → Toggle like (like si pas liké, unlike si déjà liké)
       → Met à jour likes_count dans nh_posts
       → Retourne : { liked: true/false, count: X }

-- COMMENTAIRES --
GET    /api/feed/:id/comments
       → Tous les commentaires d'un post
       → Inclure auteur + date

POST   /api/feed/:id/comment
       → Body : { content, parent_id (optionnel) }
       → Met à jour comments_count dans nh_posts
       → Envoie notification à l'auteur du post

-- PARTAGE LINKEDIN --
POST   /api/feed/:id/share-linkedin
       → Vérifie que l'utilisateur a un 
         linkedin_access_token en DB
       → Appelle LinkedIn UGC Posts API
       → Met à jour shared_to_linkedin = true
       → Sauvegarde linkedin_post_id

-- Q&A SUR LES OFFRES --
GET    /api/jobs/:id/questions
       → Questions publiques sur cette offre

POST   /api/jobs/:id/questions
       → Candidat pose une question
       → Notifie l'employeur

POST   /api/jobs/:id/questions/:qid/answer
       → Seulement l'employeur de ce job
       → Body : { answer }
       → Met à jour answered_at
