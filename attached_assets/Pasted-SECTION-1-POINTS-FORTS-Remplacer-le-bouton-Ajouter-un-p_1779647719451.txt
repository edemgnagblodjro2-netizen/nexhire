JS :
let highlights = []

async function loadHighlights() {
  const res = await fetch('/api/profile/highlights')
  highlights = await res.json()
  renderHighlights()
}

function renderHighlights() {
  const grid = document.getElementById('highlights-grid')
  grid.innerHTML = highlights.map((h, i) => `
    <div class="highlight-card">
      <button class="highlight-remove" 
              data-action="remove-highlight" 
              data-id="${h.id}">✕</button>
      <div class="highlight-icon">${h.icon}</div>
      <div class="highlight-title">${h.title}</div>
      <div class="highlight-desc">${h.description}</div>
      ${h.link ? `<a href="${h.link}" target="_blank" 
                    class="highlight-link">Voir →</a>` : ''}
    </div>
  `).join('')
}

async function saveHighlight() {
  const highlight = {
    icon: selectedIcon || '⭐',
    title: document.getElementById('hl-title').value.trim(),
    description: document.getElementById('hl-desc').value.trim(),
    link: document.getElementById('hl-link').value.trim()
  }
  if (!highlight.title) return

  await fetch('/api/profile/highlights', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(highlight)
  })
  
  closeModal('modal-highlight')
  loadHighlights()
}

function renderSkills(skills) {
  // Hard skills
  const hardWrap = document.getElementById('hard-skills-list')
  const hard = skills.filter(s => s.type === 'hard')
  hardWrap.innerHTML = hard.map(s => `
    <div class="skill-bar-wrap">
      <div class="skill-bar-header">
        <span>${s.name}</span>
        <span class="skill-level-label">
          ${s.level >= 85 ? 'Expert' : 
            s.level >= 65 ? 'Avancé' : 
            s.level >= 40 ? 'Intermédiaire' : 'Débutant'}
          · ${s.level}%
        </span>
      </div>
      <div class="skill-bar-track">
        <div class="skill-bar-fill" style="width: ${s.level}%"></div>
      </div>
    </div>
  `).join('')

  // Soft skills
  const softWrap = document.getElementById('soft-skills-list')
  softWrap.innerHTML = `<div class="soft-skills-wrap">` +
    skills.filter(s => s.type === 'soft').map(s => `
      <div class="skill-tag">
        ${s.name}
        <span class="remove-skill" 
              data-action="remove-skill" 
              data-id="${s.id}">✕</span>
      </div>
    `).join('') + `</div>`
}

SECTION 3 — ROUTES API à créer dans server.js

// Points forts
GET    /api/profile/highlights     → retourne tous les highlights du user
POST   /api/profile/highlights     → ajoute un highlight
DELETE /api/profile/highlights/:id → supprime un highlight

// Compétences  
GET    /api/profile/skills         → retourne toutes les skills
POST   /api/profile/skills         → ajoute une skill {name, level, type}
DELETE /api/profile/skills/:id     → supprime une skill
PUT    /api/profile/skills/:id     → update le niveau

// Tables DB à créer :
CREATE TABLE IF NOT EXISTS highlights (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  icon TEXT DEFAULT '⭐',
  title TEXT NOT NULL,
  description TEXT,
  link TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
)

CREATE TABLE IF NOT EXISTS skills (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  name TEXT NOT NULL,
  level INTEGER DEFAULT 75,
  type TEXT DEFAULT 'hard',
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
)
