const fs = require('fs')
const pdfParse = require('pdf-parse')

async function extractTextFromCV(filePath) {
  try {
    // Vérifier que le fichier existe
    if (!fs.existsSync(filePath)) {
      throw new Error('Fichier introuvable : ' + filePath)
    }

    const fileBuffer = fs.readFileSync(filePath)
    const extension = filePath.split('.').pop().toLowerCase()

    if (extension === 'pdf') {
      // ✅ Fix : appel correct de pdf-parse
      const data = await pdfParse(fileBuffer)
      return data.text

    } else if (extension === 'docx') {
      // Pour les .docx utiliser mammoth
      const mammoth = require('mammoth')
      const result = await mammoth.extractRawText({ path: filePath })
      return result.value

    } else {
      throw new Error('Format non supporté. Utiliser PDF ou DOCX.')
    }

  } catch (err) {
    console.error('Erreur extraction CV:', err.message)
    throw err
  }
}

// Dans server.js :
const PDFParser = require('pdf2json')

async function extractTextFromPDF(filePath) {
  return new Promise((resolve, reject) => {
    const pdfParser = new PDFParser()
    
    pdfParser.on('pdfParser_dataReady', (pdfData) => {
      const text = pdfParser.getRawTextContent()
      resolve(text)
    })
    
    pdfParser.on('pdfParser_dataError', (err) => {
      reject(err)
    })
    
    pdfParser.loadPDF(filePath)
  })
}
