import re
def extract_postal(addr):
    if pd.isna(addr): return None
    m = re.search(r'\b([A-Za-z]\d[A-Za-z])\s*(\d[A-Za-z]\d)\b', str(addr))
    if m: return f"{m.group(1).upper()} {m.group(2).upper()}"
    return None
